
# Minimal install: everything EXCEPT the cross-encoder reranker.
#
# Use this file instead of requirements.txt if PyTorch will not install or
# will not load on your machine - old CPUs, locked-down work laptops, or the
# classic Windows error:
#
#   OSError: [WinError 1114] A dynamic link library (DLL) initialization
#   routine failed. Error loading "...torch\lib\c10.dll"
#
# Install:
#   pip install -r requirements-minimal.txt
#
# Then set this line in your .env file:
#   RERANKER=simple

# multi-query, parent-child, context refinement, citations, memory, the

# cross-encoder model to simple keyword overlap.
